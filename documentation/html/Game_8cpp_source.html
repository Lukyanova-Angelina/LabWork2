142// Handles player input and movement
143void Game::handleInput(char input) {
144 std::map<char, int> directionMap = {{'w', 0}, {'d', 1}, {'s', 2}, {'a', 3}};
145 auto it = directionMap.find(input);
146 if (it != directionMap.end()) {
147
148 int direction = it->second;
149 if (player->possibletomove(direction)) {
150 int poswheremove = player->getTargetPosition(direction);
151 if (!game[poswheremove]) return;
152 int playerpos = player->getPosition();
153 switch(game[poswheremove]->returnInteractionType()){
154 case InteractionType::IMMEDIATE_PASS:
155 handleIMMEDIATE_PASS(playerpos, poswheremove);
156 break;
157 case InteractionType::SWAP_REQUIRED:
158 handleSWAP_REQUIRED(playerpos, poswheremove);
159 break;
160 case InteractionType::TRIGGER_ON_STAY:
161 handleTRIGGER_ON_STAY(poswheremove);
162 break;
163 case InteractionType::COMBAT_DEPENDENT:
164 handleCOMBAT_DEPENDENT(playerpos, poswheremove);
165 break;
166 }
167
168 update();
169 }
170 } else {
171 std::cout << "Unknown key: " << input << std::endl;
172 }

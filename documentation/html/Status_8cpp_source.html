177ObjectType Status::RandomSelection() const {
178 if (LootTable.empty()) {
179 std::cerr << "ERROR: LootTable is empty! Returning RUBY as fallback." << std::endl;
180 return ObjectType::OBJECT;
181 }
182 double totalWeight = 0.0;
183 for (const auto& item : LootTable) {
184 totalWeight += item.second;
185 }
186 double randomValue = getRandomValue() * totalWeight;
187 double sumWeight = 0.0;
188
189 for (const auto& item : LootTable){
190 sumWeight += item.second;
191 if (randomValue <= sumWeight + 1e-10){
192 return item.first;
193 }
194 }
195 return LootTable.rbegin()->first;
196}

150std::string centerText(const std::string& text, size_t width) {
151 if (text.empty()) {
152 return std::string(width, ' ');
153 }
154
155
156 static const std::unordered_map<std::string, int> symbolCorrections = {
157 {"→", -2},
158 {"←", -2},
159 {"↓", -2},
160 {"↑", -2},
161 {"▲", -2},
162 {"◀", -2},
163 {"▶", -2},
164 {"▼", -2},
165 {"⚔", -2}
166 };
167
168 size_t effectiveLength = text.length();
169
170 for (const auto& [symbol, correction] : symbolCorrections) {
171 if (text.find(symbol) != std::string::npos) {
172 effectiveLength += correction;
173 }
174 }
175
176 if (effectiveLength >= width) {
177 return text.substr(0, width);
178 }
179
180 size_t leftPadding = (width - effectiveLength) / 2;
181 size_t rightPadding = width - effectiveLength - leftPadding;
182
183 return std::string(leftPadding, ' ') + text + std::string(rightPadding, ' ');
184}
